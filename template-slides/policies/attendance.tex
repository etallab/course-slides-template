\begin{frame}
  \frametitle{Attendance}

  \begin{itemize}
  \item You are encouraged to attend class when possible.
  \item Any attendance after the second week is worth 2\% extra credit (on top of your grade).
  \item If you are unwell or traveling on university business (sports, \etc.), you may choose to attend by Teams session.
    You must notify me of your choice at least 30 minutes in advance of class.
  \item[\(\Rightarrow\)] You must complete the Syllabus Quiz by the census date, or you will not be considered to have attended.
  \end{itemize}
  
\end{frame}
