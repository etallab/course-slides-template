\begin{frame}
  \frametitle{Generative AI}

  \begin{itemize}
  \item[\(\rightarrow\)] Unless \emph{explicitly} stated, Generative AI tools are forbidden.
  \item[\(\rightarrow\)] Use of Generative AI tools unless permitted is an academic integrity violation.
    \vspace{1em}
  \item When Generative AI is permitted, the following conditions must be met:
    \begin{enumerate}
    \item Details of context and prompt must be provided;
    \item Tool provider, model, and temperature setting are listed; and,
    \item You must detail why Generative AI was chosen over other resources.
    \end{enumerate}
  \end{itemize}
  
\end{frame}
