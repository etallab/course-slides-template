\begin{frame}
  \frametitle{Academic Integrity}

  \begin{itemize}
  \item[\(\rightarrow\)] Violations of Academic Integrity are defined by SDBOR Policy 3.4.1, section 2.1.
  \item[\(\rightarrow\)] Student Conduct Policies of SDBOR Policy 2.9.2, and from there SDBOR Policy 3.4.1 section 3.2 are applied
    \vspace{1em}
  \item \textbf{Academic Penalties for Undergrads:} (Previous instances considered)
    \begin{itemize}
    \item First instance: Failing grade on the assignment.
    \item Second instance: Failing grade on the course.
    \item Third instance or beyond: No informal resolution possible.
    \end{itemize}
  \item \textbf{Academic Penalties for Grads:} A failing grade in this course.
  \end{itemize}
  
\end{frame}
